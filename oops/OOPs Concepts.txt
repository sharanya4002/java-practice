	{
		this.Accnum = Accnum;

import java.util.Scanner;
class Pay
{
	void pay()
	{
		System.out.print("Making Payment");
	}
}
class Upi extends Pay
{
	void pay()
	{
		System.out.print("payed using upi");
	}
}
class CreditCard extends Pay
{
	void pay()
	{
		System.out.print("payed using creditcard");
	}
}
class DebitCard extends Pay
{
	void pay()
	{
		System.out.print("payed using debitcard");
	}
}
class NetBanking extends Pay
{
	void pay()
	{
		System.out.print("payed using netbanking");
	}
}
class Main
{
	static void makePayment(Pay p)
	{
		p.pay();
	}
	public static void main(String args[])
	{
		Scanner sc = new Scanner(System.in);
		Upi upi = new Upi();
		CreditCard cc= new CreditCard();
		DebitCard dc = new DebitCard();
		NetBanking nb = new NetBanking();
		System.out.println("Enter your Payment type");
		System.out.printf("1.upi\n2.credit card\n3.netbanking\n4.debit card\n");
		int n = sc.nextInt();
		switch(n)
		{
			case 1: makePayment(upi); break;
			case 2: makePayment(cc); break;
			case 3: makePayment(nb); break;
			case 4: makePayment(dc); break;
		}
	}
}
