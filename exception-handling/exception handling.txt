{

5.Create a java application where we need to deal with exception classes.  (haha.java)

import java.util.Scanner;

class InvalidAgeException extends Exception
{
    InvalidAgeException(String msg)
    {
        super(msg);
    }
}

class A
{
    static void checkAge(int age) throws InvalidAgeException
    {
        if(age < 18)
        {
            throw new InvalidAgeException("Age must be 18 or above");
        }
        else
        {
            System.out.println("Eligible for voting");
        }
    }

    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);

        try
        {
            System.out.print("Enter age: ");
            int age = sc.nextInt();

            checkAge(age);
        }
        catch(InvalidAgeException e)
        {
            System.out.println("Custom Exception: " + e.getMessage());
        }
        catch(Exception e)
        {
            System.out.println("Exception: " + e);
        }
        finally
        {
            System.out.println("Program executed successfully");
        }
    }
}

7.Create a Java application where we need to create custom-defined exceptions like InvalidLoginException, need to create a Gmail application by validating username and password, if we give valid credentials we need to get a welcome message, if we give invalid credentials then that application has to raise custom defined exception like invalidLoginException with respect to a user-friendly error message by using both throws and throw keyword

import java.util.Scanner;
class InvalidLoginException extends Exception
{
	InvalidLoginException(String msg)
	{
		super(msg);
	}
}
class Gmail
{
	private String username;
	private String password;
	Gmail(String username,String password)
	{
		this.username = username;
		this.password = password;
	}
	String getUname()
	{
		return username;
	}
	void setUname(String uname)
	{
		this.username = uname;
	}
	String getPassword()
	{
		return password;
	}
	void setPassword(String password)
	{
		this.password = password;
	}
}
class Main
{
	static Scanner sc = new Scanner(System.in);
	static Gmail obj =new  Gmail(sc.next(),sc.next());
	public static void validateCredentials(String uname,String password) throws InvalidLoginException
	{
		if(uname.equals(obj.getUname()) && password.equals(obj.getPassword()))
		{
			System.out.print("login succesful welcome");
		}
		else
		{
			throw new InvalidLoginException("Invalid Credentials");
		}
	}
		
	public static void main(String args[])
	{
		Scanner sc = new Scanner(System.in);
		System.out.print("enter credentials to validate");
		try
		{
			validateCredentials(sc.next(),sc.next());
		}
		catch(Exception e)
		{
			e.printStackTrace();
		}

	}
}
