1.	create java application where we have one class that contains one private variable and one method with parameter and return type, inside this class, we have two inner classes, which contain one method with parameter and return type individually, these methods have to access the private variable and then we need to invoke all these methods under main by providing dynamic inputs (innerclass.java)

import java.util.*;
class A
{
	static Scanner sc=new Scanner(System.in);
	private String name;
	A(String name)
	{
		this.name=name;
	}
	boolean m1(int n)
	{
		System.out.println(n);
		return sc.nextBoolean();
	}

	class B1
	{
		String m2(boolean b)
		{
			System.out.println(name);
			System.out.println(b);
			return sc.next();
		}

	}

	class B2
	{
		int m3(float f)
		{
			System.out.println(name);
			System.out.println(f);
			return sc.nextInt();
		}
	}
}

class Main
{
	public static void main(String args[])
	{
		Scanner sc=new Scanner(System.in);
		A obj=new A(sc.next());
		System.out.println(obj.m1(sc.nextInt()));
		A.B1 o1=obj.new B1();
		System.out.println(o1.m2(sc.nextBoolean()));
		A.B2 o2=obj.new B2();
		System.out.println(o2.m3(sc.nextFloat()));
	}
}

7.	create a java application where we have one interface it contains one abstract method and one defined method, inside this interface we have two interfaces with respect to individual abstract methods, then provide an implementation for all these interfaces into one concrete class and then invoke all these methods by providing dynamic inputs. (nested.java)

import java.util.*;

interface I
{
    boolean m1(int n);

    static void m2(String s)
    {
        System.out.println(s);
    }

    interface I1
    {
        String m3(float f);
    }

    interface I2
    {
        double m4(int d);
    }
}

class A implements I, I.I1, I.I2
{
    static Scanner sc = new Scanner(System.in);

    public boolean m1(int n)
    {
        System.out.println(n);
        return sc.nextBoolean();
    }

    public String m3(float f)
    {
        System.out.println(f);
        return sc.next();
    }

    public double m4(int d)
    {
        System.out.println(d);
        return sc.nextDouble();
    }
}

class Main
{
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);

        A obj = new A();

        System.out.println(obj.m1(sc.nextInt()));

        I.m2(sc.next());

        System.out.println(obj.m3(sc.nextFloat()));

        System.out.println(obj.m4(sc.nextInt()));
    }
}				
			
8.	create a java application where we have one class it contains one parameterized constructor, one method with a parameter, and a return type, inside this class we have one interface it contains one abstract method and two defined methods, provide an implementation for the interface and then invoke all the properties under main by providing dynamic inputs. (CI.java)

import java.util.*;
class A
{
	static Scanner sc=new Scanner(System.in);
	A(String s)
	{
		System.out.println(s);
	}

	boolean m1(int n)
	{
		System.out.println(n);
		return sc.nextBoolean();
	}


	interface I
	{
		Scanner sc=new Scanner(System.in);
		float m2(boolean b);
		
		static int m3(float f)
		{

		}

		default double m4(int g)
		{
			System.out.println(g);
			return sc.nextDouble();
		}
	}

import java.util.*;
interface I
{
	static Scanner sc=new Scanner(System.in);
	boolean m1(int n);
	static void m2(String s)
	{
		System.out.println(s);
	}

	class A
	{
		int m3(int d)
		{
			System.out.println(d);
			return sc.nextInt();
		}
	}

	class B
	{
		float m4(float f)
		{
			System.out.print(f);
			return sc.nextFloat();
		}
	}
}

class C implements I
{
	public boolean m1(int n)
	{
		System.out.println(n);
		return I.sc.nextBoolean();
	}
}

class Main
{
	public static void main(String args[])
	{
		Scanner sc=new Scanner(System.in);
		C obj=new C();
		System.out.println(obj.m1(sc.nextInt()));

		I.A obj1=new I.A();
		System.out.println(obj1.m3(sc.nextInt()));

		System.out.println(obj2.m4(sc.nextFloat()));
