import java.util.Scanner;
class A implements Runnable
{
	public void run()
	{
		Thread t = Thread.currentThread();
		String s = t.getName();
		if(s.equals("t1"))
			printmsg();
		else if(s.equals("t2"))
			printNumbers();
		else
			printChar();
	}
	void printmsg()
	{
		System.out.println("welcome"+Thread.currentThread().getName());
	}
	void printNumbers()
	{
		for(int i=0;i<10;i++)
		{
			System.out.println(i+" "+Thread.currentThread().getName());
		}
	}
	void printChar()
	{
		for(char c ='A';c<'G';c++)
		{
			System.out.println(c+" "+Thread.currentThread().getName());
		}
	}
	public static void main(String args[])
	{
		A obj = new A();
		Thread t1 = new Thread(obj);
		t1.setName("t1");
		Thread t2 = new Thread(obj);
		t2.setName("t2");
		Thread t3 = new Thread(obj);
		t3.setName("t3");
		t1.start();
		t2.start();
		t3.start();
		
	}
}
-------------------------------Daemon Thread--------------------------------------------
class Eclipse implements Runnable
{
	public void run()
	{
		Thread t = Thread.currentThread();
		String name = t.getName();
		if(name.equals("typing"))
			typing();
		else if(name.equals("saved"))
			autoSave();
		else
			autoCompile();
		
	}
	void typing()
	{
		for(int i=0;i<10;i++)
		{
			System.out.println("typing..............");
			try
			{
				Thread.sleep(1000);
			}
			catch(Exception e)
			{
				System.out.print("error");
			}
			
		}
	}
	void autoSave()
	{
		// this for loop terminates even without the termination condition as thread used to call this method is daemon thread  and daemon thread depends on user thread i.e current thread 
		for(int i=0;;i++)
		{
			System.out.println("saved............");
			try
			{
				Thread.sleep(1000);
			}
			catch(Exception e)
			{
				System.out.print("error");
			}


		}
	}
	void autoCompile()
	{
		// this for loop terminates even without the termination condition as thread used to call this method is daemon thread  and daemon thread depends on user thread i.e current thread 
		for(int i=0;;i++)
		{
			System.out.println("compiling............");
			try
			{
				Thread.sleep(1000);
			}
			catch(Exception e)
			{
				System.out.print("error");
			}

		}
	}
	public static void main(String args[])
	{
		Eclipse e = new Eclipse();
		Thread t1 = new Thread(e);
		Thread t2 = new Thread(e);
		Thread t3 = new Thread(e);
		
		t1.setName("typing");
		t2.setName("saved");
		t3.setName("compiling");
		
		//creation of daemon thread
		t2.setDaemon(true);
		t3.setDaemon(true);
		
		t1.start();
		t2.start();
		t3.start();
	}

}
